-- Drop existing tables si quieres recrearlas
DROP TABLE IF EXISTS application CASCADE;
DROP TABLE IF EXISTS loan_types CASCADE;
DROP TABLE IF EXISTS states CASCADE;

-- Crear estados con INTEGER
CREATE TABLE IF NOT EXISTS states (
    state_id INTEGER PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL,
    description VARCHAR(200)
);

INSERT INTO states (state_id, name, description) VALUES
    (1, 'PENDING_REVIEW', 'Application is pending review'),
    (2, 'APPROVED', 'Application has been approved'),
    (3, 'REJECTED', 'Application has been rejected')
ON CONFLICT (state_id) DO NOTHING;

-- Crear tipos de préstamo con INTEGER
CREATE TABLE IF NOT EXISTS loan_types (
    loan_type_id INTEGER PRIMARY KEY,
    name VARCHAR(50) UNIQUE NOT NULL,
    minimum_amount NUMERIC(15, 2) NOT NULL,
    maximum_amount NUMERIC(15, 2) NOT NULL,
    interest_rate NUMERIC(5, 4) NOT NULL,
    automatic_validation BOOLEAN NOT NULL
);

INSERT INTO loan_types (loan_type_id, name, minimum_amount, maximum_amount, interest_rate, automatic_validation) VALUES
    (1, 'PERSONAL', 500.00, 20000.00, 0.2000, FALSE),
    (2, 'VEHICLE', 5000.00, 80000.00, 0.1200, FALSE),
    (3, 'HOUSING', 30000.00, 500000.00, 0.0800, FALSE),
    (4, 'EDUCATIONAL', 1000.00, 50000.00, 0.0500, TRUE)
ON CONFLICT (loan_type_id) DO NOTHING;

-- Crear tabla de aplicaciones
CREATE TABLE IF NOT EXISTS application (
    application_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    client_document VARCHAR(50) NOT NULL,
    amount NUMERIC(15, 2) NOT NULL,
    term_months INTEGER NOT NULL,
    email VARCHAR(150) NOT NULL,
    state_id INTEGER NOT NULL,
    loan_type_id INTEGER NOT NULL,
    created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT fk_state
        FOREIGN KEY (state_id)
        REFERENCES states(state_id),
    CONSTRAINT fk_loan_type
        FOREIGN KEY (loan_type_id)
        REFERENCES loan_types(loan_type_id)
);
